-- Zadanie 1.4

SELECT Name, AnnualIncome, AnnualIncome / 12 AS 'Przychód miesięczny' 
FROM Company 
WHERE Active = 1 AND EmployeeCount > 1000;


-- Zadanie 1.5

FROM Company 
ORDER BY AnnualIncome / 12 DESC 
LIMIT 3;


-- Zadanie 1.6

-- Zadanie 1.7

-- Sposób 1

FROM Company 
WHERE PostCode IN ('54-621', '78-695', '63-896');
-- Sposób 2
SELECT Name, PostCode, CONCAT(Street, ' ', Number) AS Adres 
FROM Company 
WHERE PostCode = '54-621' OR PostCode = '78-695' OR PostCode = '63-896';


-- Zadanie 1.8
