-- Ćwiczenie 2.1
-- Dla każdego pracownika wyświetl imię, nazwisko oraz rok jego urodzenia
SELECT FirstName, LastName, YEAR(BirthDate) AS 'Rok urodzenia'
FROM employee;

-- Ćwiczenie 2.2
-- Dla każdego pracownika wyświetl imię, nazwisko oraz rok, miesiąc i dzień jego urodzenia 
-- oraz dzień tygodnia, w jakim się urodził
SELECT 
    FirstName, 
    LastName, 
    YEAR(BirthDate) AS 'Rok urodzenia',
    MONTH(BirthDate) AS 'Miesiąc',
    DAY(BirthDate) AS 'Dzień',
    DAYNAME(BirthDate) AS 'Dzień tygodnia'
FROM employee;

-- Ćwiczenie 2.3
-- Wyświetl wszystkich pracowników urodzonych w 1990 roku
SELECT * 
FROM employee 
WHERE YEAR(BirthDate) = 1990;

-- Ćwiczenie 2.4
-- Wyświetl wszystkich pracowników urodzonych w maju 1990 roku
SELECT * 
FROM employee 
WHERE YEAR(BirthDate) = 1990 AND MONTH(BirthDate) = 5;

-- Ćwiczenie 2.5
-- Wyświetl bieżącą datę i godzinę
SELECT NOW() AS 'Teraz';

-- Ćwiczenie 2.6
-- Wyświetl, ile lat minęło od daty 1 stycznia 1990 roku
SELECT TIMESTAMPDIFF(YEAR, '1990-01-01', NOW()) AS 'IleLat';

-- Ćwiczenie 2.7
-- Dla każdego pracownika wyświetl jego imię, nazwisko oraz wiek
SELECT 
    FirstName, 
    LastName, 
    TIMESTAMPDIFF(YEAR, BirthDate, NOW()) AS 'Wiek'
FROM employee;

-- Ćwiczenie 2.8
-- Wyświetl wszystkich pracowników, którzy mają więcej niż 25 lat
SELECT * 
FROM employee 
WHERE TIMESTAMPDIFF(YEAR, BirthDate, NOW()) > 25;

-- Ćwiczenie 2.9
-- Wyświetl dane wszystkich pracowników, którzy mają mniej niż 30 lat i zarabiają mniej niż 2000 zł
SELECT * 
FROM employee 
WHERE TIMESTAMPDIFF(YEAR, BirthDate, NOW()) < 30 AND Salary < 2000;

-- Ćwiczenie 2.10
-- Dla każdego aktywnego pracownika wyświetl imię, nazwisko oraz informację, ile lat już pracuje w tej firmie
SELECT 
    FirstName, 
    LastName, 
    TIMESTAMPDIFF(YEAR, HireDate, NOW()) AS 'LiczbąLat'
FROM employee 
WHERE Active = 1;

-- Ćwiczenie 2.11
-- Dla każdego aktywnego pracownika wyświetl imię, nazwisko oraz informację, ile miesięcy już pracuje w tej firmie
SELECT 
    FirstName, 
    LastName, 
    TIMESTAMPDIFF(MONTH, HireDate, NOW()) AS 'LiczbąMiesięcy'
FROM employee 
WHERE Active = 1;

-- Ćwiczenie 2.12
-- Wyświetl dane wszystkich pracowników, którzy przepracowali w tej firmie co najmniej 10 miesięcy, 
-- ale nie więcej niż 20
SELECT * 
FROM employee 
WHERE TIMESTAMPDIFF(MONTH, HireDate, NOW()) >= 10 
  AND TIMESTAMPDIFF(MONTH, HireDate, NOW()) <= 20;

-- Ćwiczenie 2.13
-- Dla każdego pracownika wyświetl imię, nazwisko, rok urodzenia, rok, w którym został zatrudniony, 
-- oraz liczbę lat, którą miał w momencie zatrudnienia
SELECT 
    FirstName AS 'Imię', 
    LastName AS 'Nazwisko', 
    YEAR(BirthDate) AS 'Rok urodzenia',
    YEAR(HireDate) AS 'Rok zatrudnienia',
    TIMESTAMPDIFF(YEAR, BirthDate, HireDate) AS 'Wiek w momencie zatrudnienia'
FROM employee;

-- Ćwiczenie 2.14
-- Sprawdź, czy istnieją pracownicy, którzy przez pomyłkę mają wpisaną datę zatrudnienia 
-- późniejszą niż datę zwolnienia
SELECT * 
FROM employee 
WHERE HireDate > RelieveDate;

-- Ćwiczenie 2.15
-- Dla każdego byłego pracownika, który nie pracuje już w firmie, wyświetl imię i nazwisko 
-- oraz liczbę miesięcy, którą przepracował w firmie
SELECT 
    FirstName AS 'Imię', 
    LastName AS 'Nazwisko', 
    TIMESTAMPDIFF(MONTH, HireDate, RelieveDate) AS 'Liczba przepracowanych miesięcy'
FROM employee 
WHERE Active = 0 AND RelieveDate IS NOT NULL;

-- Ćwiczenie 2.16
-- Dla każdego pracownika wyświetl imię, nazwisko oraz inicjały
SELECT 
    FirstName AS 'Imię', 
    LastName AS 'Nazwisko', 
    LEFT(FirstName, 1) AS 'Pierwsza litera imienia',
    LEFT(LastName, 1) AS 'Pierwsza litera nazwiska'
FROM employee;

-- Ćwiczenie 2.17
-- Dla każdego pracownika wyświetl nazwisko oraz liczbę liter w nazwisku
SELECT 
    LastName AS 'Nazwisko', 
    LENGTH(LastName) AS 'Liczba liter w nazwisku'
FROM employee;

-- Ćwiczenie 2.18
-- Wyświetl wszystkich pracowników, których nazwisko jest dłuższe niż 6 liter
SELECT * 
FROM employee 
WHERE LENGTH(LastName) > 6;

-- Ćwiczenie 2.19
-- Dla każdego pracownika wyświetl jego imię, ale pisane małymi literami, oraz nazwisko pisane wielkimi literami
SELECT 
    LOWER(FirstName) AS 'Imię', 
    UPPER(LastName) AS 'Nazwisko'
FROM employee;

-- Ćwiczenie 2.20
-- Dla każdego pracownika wyświetl jego login. Login jest zbudowany z pierwszej litery imienia i całego nazwiska
SELECT 
    CONCAT(LEFT(FirstName, 1), LastName) AS 'Login'
FROM employee;

-- Ćwiczenie 2.21
-- Dla każdego aktywnego pracownika wyświetl jego login napisany małymi literami. 
-- Login, jak poprzednio, jest zbudowany z pierwszej litery imienia i całego nazwiska
SELECT 
    LOWER(CONCAT(LEFT(FirstName, 1), LastName)) AS 'Login'
FROM employee 
WHERE Active = 1;

-- Ćwiczenie 2.22
-- Dla każdego pracownika wyświetl krótki login, który jest zbudowany z pierwszej litery imienia 
-- oraz pięciu pierwszych liter nazwiska. Login ten wyświetl przy użyciu wielkich liter. 
-- Przykładowo, dla pracownika John Williams krótki login to JWILLI
SELECT 
    UPPER(CONCAT(LEFT(FirstName, 1), LEFT(LastName, 5))) AS 'Krótki login'
FROM employee;

-- Ćwiczenie 2.23
-- Wyświetl imię, nazwisko oraz wiek wszystkich pracowników. Posortuj ich dane według wieku, 
-- a jeśli istnieje wielu pracowników w tym samym wieku, posortuj ich według nazwiska. 
-- Jeśli istnieje wielu pracowników w tym samym wieku i o tym samym nazwisku, posortuj ich według imienia
SELECT 
    FirstName AS 'Imię', 
    LastName AS 'Nazwisko', 
    TIMESTAMPDIFF(YEAR, BirthDate, NOW()) AS 'Data urodzenia'
FROM employee 
ORDER BY TIMESTAMPDIFF(YEAR, BirthDate, NOW()), LastName, FirstName;
